\section{Overview}

Modal adjacency defines the semantic transition structure of the reasoning
manifold. Each point $p \in M$ has a modal label $\mu(p)$, and transitions
between points must respect a fixed adjacency graph. Modal violations are
a primary collapse trigger and contribute directly to holonomy drift and
stability envelopes.

This module depends on \texttt{core-math.tex} and \texttt{collapse-geometry.tex}.

\section{Modal Graph}

Let $L$ be a finite set of modal labels. Define the modal adjacency graph:
\[
G = (L, E),
\]
where $E \subseteq L \times L$ is a directed edge set.

An edge $(a,b) \in E$ indicates that a transition from mode $a$ to mode $b$
is semantically valid.

\section{Modal Label}

Each point $p \in M$ has a modal label:
\[
\mu(p) \in L.
\]

\section{Valid Modal Transition}

A transition $p^- \to p^+$ is valid when:
\[
(\mu(p^-), \mu(p^+)) \in E.
\]

\subsection{Lemma: Modal Validity Implies Semantic Stability}
If $(\mu(p^-), \mu(p^+)) \in E$, then the transition is semantically stable.

\begin{proof}
Direct from definition of adjacency.
\end{proof}

\section{Modal Violation}

A modal violation occurs when:
\[
(\mu(p^-), \mu(p^+)) \notin E.
\]

Define the modal collapse predicate:
\[
C_{\mathrm{modal}}(p)
  := (\mu(p^-), \mu(p^+)) \notin E.
\]

\subsection{Lemma: Modal Violation Implies Collapse}
If $C_{\mathrm{modal}}(p)$ holds, then $p \in \Sigma$.

\begin{proof}
Modal adjacency is a structural invariant; violation triggers collapse.
\end{proof}

\section{Modal Distance}

Define modal distance between labels $a,b \in L$:
\[
d_{\mathrm{modal}}(a,b)
  = \min_{\gamma : a \to b}
      |\gamma|,
\]
where $\gamma$ ranges over all paths in $G$.

Modal distance is used in drift accumulation.

\section{Modal Drift}

Define modal drift along a path $\gamma = (p_0, \ldots, p_n)$:
\[
D_{\mathrm{modal}}(\gamma)
  = \sum_{i=0}^{n}
      d_{\mathrm{modal}}(\mu(p_i), \mu(p_{i+1})).
\]

Modal drift contributes directly to holonomy drift.

\section{Modal Stability Condition}

A point $p$ is modally stable when:
\[
d_{\mathrm{modal}}(\mu(p^-), \mu(p^+)) = 1.
\]

This ensures transitions follow adjacency edges.

\section{Modal Contribution to Holonomy Drift}

Define the modal contribution to holonomy drift:
\[
H_{\mathrm{modal}}(p)
  = d_{\mathrm{modal}}(\mu(p^-), \mu(p^+)).
\]

This term is used in the full holonomy drift tensor.

\section{Modal Envelope}

Define the modal envelope:
\[
E_{\mathrm{modal}}(p)
  = \{ q \in M : d_{\mathrm{modal}}(\mu(p), \mu(q)) \le \delta_{\mathrm{modal}} \}.
\]

Points outside the modal envelope are near-collapse.
