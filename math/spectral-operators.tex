\section{Overview}

Spectral operators describe how local linearizations of reasoning behave
under drift, curvature, and modal transitions. Spectral mismatch is one of
the primary collapse triggers and contributes directly to holonomy drift
and stability envelopes.

This module depends on \texttt{core-math.tex} and \texttt{collapse-geometry.tex}.

\section{Local Operator}

Each operator field induces a local linear operator:
\[
L(p) : T_pM \to T_pM.
\]

We assume $L(p)$ is diagonalizable almost everywhere, with eigenvalues:
\[
\mathrm{Spec}(L(p)) = \{\lambda_1(p), \ldots, \lambda_n(p)\}.
\]

\section{Spectral Continuity}

Spectral continuity requires:
\[
\lim_{q \to p}
  \|\mathrm{Spec}(L(q)) - \mathrm{Spec}(L(p))\| = 0.
\]

\subsection{Lemma: Spectral Continuity Implies Operator Stability}
If spectral continuity holds at $p$, then $L(p)$ is stable under infinitesimal
drift.

\begin{proof}
Eigenvalues vary continuously under smooth perturbations.
\end{proof}

\section{Expected Spectrum}

Define the expected spectrum:
\[
\mathrm{Spec}_{\mathrm{expected}}(p)
  = f(K(p), \mu(p), X(p)),
\]
where $f$ is a deterministic map derived from curvature, modal label, and drift.

This map is fixed by the deterministic kernel.

\section{Spectral Mismatch}

Spectral mismatch is defined as:
\[
\delta_{\mathrm{spec}}(p)
  = \left\|
      \mathrm{Spec}(L(p))
      - \mathrm{Spec}_{\mathrm{expected}}(p)
    \right\|.
\]

\subsection{Mismatch Threshold}
A point is spectrally stable when:
\[
\delta_{\mathrm{spec}}(p) < \theta_{\mathrm{spec}}.
\]

\section{Spectral Collapse Predicate}

Spectral collapse occurs when:
\[
C_{\mathrm{spec}}(p)
  := \delta_{\mathrm{spec}}(p) \ge \theta_{\mathrm{spec}}.
\]

\subsection{Lemma: Spectral Collapse Implies $p \in \Sigma$}
If $C_{\mathrm{spec}}(p)$ holds, then $p$ lies on the collapse surface.

\begin{proof}
Direct from collapse predicate definition.
\end{proof}

\section{Spectral Drift}

Define spectral drift along a path $\gamma = (p_0, \ldots, p_n)$:
\[
D_{\mathrm{spec}}(\gamma)
  = \sum_{i=0}^{n}
      \delta_{\mathrm{spec}}(p_i).
\]

Spectral drift contributes directly to holonomy drift.

\section{Spectral Curvature Interaction}

Define the spectral-curvature interaction term:
\[
I_{\mathrm{spec-curv}}(p)
  = |\lambda_{\max}(p)| \cdot K(p).
\]

This term is used in stability envelope inequalities.

\section{Spectral Stability Condition}

A point $p$ is spectrally stable when:
\[
\delta_{\mathrm{spec}}(p)
  + I_{\mathrm{spec-curv}}(p)
  < \theta_{\mathrm{stable}}.
\]

\section{Spectral Contribution to Holonomy Drift}

The spectral contribution to holonomy drift is:
\[
H_{\mathrm{spec}}(p)
  = \delta_{\mathrm{spec}}(p)
    + I_{\mathrm{spec-curv}}(p).
\]

This term is used in the full holonomy drift tensor.
